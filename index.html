-- Fanni Pro: admin-controlled user status and granular permissions. -- Run after backup. Review on staging first. ALTER TABLE public.user_profiles ADD COLUMN IF NOT EXISTS is_active boolean NOT NULL DEFAULT true; ALTER TABLE public.user_profiles ADD COLUMN IF NOT EXISTS permissions jsonb NOT NULL DEFAULT '{}'::jsonb; ALTER TABLE public.user_profiles ADD COLUMN IF NOT EXISTS access_updated_at timestamptz NOT NULL DEFAULT now(); CREATE OR REPLACE FUNCTION public.fp_admin_manage_user( p_user_id uuid, p_is_active boolean, p_role text, p_permissions jsonb, p_technician_id text DEFAULT NULL, p_company_id text DEFAULT NULL ) RETURNS void LANGUAGE plpgsql SECURITY DEFINER SET search_path = public, pg_temp AS $$ DECLARE v_admin uuid := auth.uid(); BEGIN IF v_admin IS NULL OR NOT EXISTS ( SELECT 1 FROM public.user_profiles WHERE id = v_admin AND role = 'admin' AND is_active = true ) THEN RAISE EXCEPTION 'Admin access required'; END IF; IF p_user_id = v_admin AND (NOT p_is_active OR p_role <> 'admin') THEN RAISE EXCEPTION 'Cannot disable or demote your own admin account'; END IF; IF p_role NOT IN ('admin','dispatcher','accountant','company_manager','technician','viewer','client') THEN RAISE EXCEPTION 'Invalid role'; END IF; IF jsonb_typeof(p_permissions) <> 'object' THEN RAISE EXCEPTION 'Permissions must be an object'; END IF; UPDATE public.user_profiles SET is_active = p_is_active, role = p_role, permissions = p_permissions, technician_id = NULLIF(p_technician_id,''), company_id = NULLIF(p_company_id,''), access_updated_at = now() WHERE id = p_user_id; IF NOT FOUND THEN RAISE EXCEPTION 'User profile not found'; END IF; END; $$; REVOKE ALL ON FUNCTION public.fp_admin_manage_user(uuid,boolean,text,jsonb,text,text) FROM PUBLIC; GRANT EXECUTE ON FUNCTION public.fp_admin_manage_user(uuid,boolean,text,jsonb,text,text) TO authenticated; -- IMPORTANT: status must also be enforced in every relevant RLS policy. -- This function can be used as a policy condition; do not replace existing -- tenant/company filters with it.
